local QuestManager = {}

QuestManager.CurrentQuest = nil


function QuestManager.SelectQuest(data)

	QuestManager.CurrentQuest = data

end


function QuestManager.StartQuest()

	local quest = QuestManager.CurrentQuest

	if not quest then
		return
	end


	local eventos = ReplicatedStorage:FindFirstChild("EventosGlobais")

	if eventos then

		local remote = eventos:FindFirstChild("DispararMissao")

		if remote then
			remote:FireServer(
				quest.Name,
				quest.Stage
			)
		end

	end

end



--==================================================
-- TRAVEL MANAGER
--==================================================

local TravelManager = {}


function TravelManager.Go(village)

	local folder = workspace:FindFirstChild("PastasDeVilas")

	if not folder then return end


	local place = folder:FindFirstChild(village)

	if not place then return end


	local spawn = place:FindFirstChild("SpawnPart")

	local char = Player.Character

	if not char then return end


	local root = char:FindFirstChild("HumanoidRootPart")

	if not root or not spawn then return end


	for _,p in pairs(char:GetDescendants()) do

		if p:IsA("BasePart") then
			p.CanCollide = false
		end

	end


	local distance =
		(root.Position - spawn.Position).Magnitude


	local tween = TweenService:Create(
		root,
		TweenInfo.new(distance / 350),
		{
			CFrame = spawn.CFrame
		}
	)


	tween:Play()

end



--==================================================
-- TARGET MANAGER
--==================================================

local TargetManager = {}


function TargetManager.Find(name)

	local folder =
		workspace:FindFirstChild("MonstrinhosDoMapa")


	if not folder then return end


	for _,enemy in pairs(folder:GetChildren()) do

		if enemy.Name == name then

			return enemy

		end

	end

end

local CombatController = {}


function CombatController.Attack()

	local char = Player.Character

	if not char then return end


	local tool =
		char:FindFirstChildOfClass("Tool")


	if tool then

		tool:Activate()

	end

end

local function Drag(obj,target)

	local dragging=false
	local start
	local pos


	obj.InputBegan:Connect(function(i)

		if i.UserInputType == Enum.UserInputType.Touch then

			dragging=true
			start=i.Position
			pos=target.Position

		end

	end)


	UserInputService.InputChanged:Connect(function(i)

		if dragging and i.UserInputType==Enum.UserInputType.Touch then

			local d=i.Position-start

			target.Position=UDim2.new(
				pos.X.Scale,
				pos.X.Offset+d.X,
				pos.Y.Scale,
				pos.Y.Offset+d.Y
			)

		end

	end)


	UserInputService.InputEnded:Connect(function()

		dragging=false

	end)

end
