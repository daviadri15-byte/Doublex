local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")

local Player = Players.LocalPlayer

local Gui = Instance.new("ScreenGui")
Gui.Name = "DOUBLEX_UI"
Gui.ResetOnSpawn = false
Gui.Parent = Player:WaitForChild("PlayerGui")

local Bubble = Instance.new("TextButton")
Bubble.Size = UDim2.new(0,60,0,60)
Bubble.Position = UDim2.new(0,20,0.5,0)
Bubble.Text = "🐺"
Bubble.TextScaled = true
Bubble.Parent = Gui

local BubbleCorner = Instance.new("UICorner")
BubbleCorner.CornerRadius = UDim.new(1,0)
BubbleCorner.Parent = Bubble

local Panel = Instance.new("Frame")
Panel.Size = UDim2.new(0,280,0,360)
Panel.Position = UDim2.new(0,100,0.3,0)
Panel.Visible = false
Panel.Parent = Gui

local PanelCorner = Instance.new("UICorner")
PanelCorner.CornerRadius = UDim.new(0,15)
PanelCorner.Parent = Panel

local Title = Instance.new("TextLabel")
Title.Size = UDim2.new(1,-50,0,45)
Title.Position = UDim2.new(0,10,0,5)
Title.Text = "🐺 DOUBLEX 🐺"
Title.TextScaled = true
Title.BackgroundTransparency = 1
Title.Parent = Panel

local Minimize = Instance.new("TextButton")
Minimize.Size = UDim2.new(0,40,0,40)
Minimize.Position = UDim2.new(1,-45,0,5)
Minimize.Text = "−"
Minimize.Parent = Panel

local Status = Instance.new("TextLabel")
Status.Size = UDim2.new(1,-20,0,35)
Status.Position = UDim2.new(0,10,0,55)
Status.Text = "Status: Pronto"
Status.TextScaled = true
Status.BackgroundTransparency = 1
Status.Parent = Panel

local function CreateButton(Text,Y)
	local Button = Instance.new("TextButton")
	Button.Size = UDim2.new(0.85,0,0,45)
	Button.Position = UDim2.new(0.075,0,0,Y)
	Button.Text = Text
	Button.TextScaled = true
	Button.Parent = Panel
	return Button
end

local QuestButton = CreateButton("📜 QUEST",110)
local AttackButton = CreateButton("⚔ ATAQUE",165)
local TravelButton = CreateButton("🚢 VIAGEM",220)
local ConfigButton = CreateButton("⚙ CONFIG",275)

Bubble.MouseButton1Click:Connect(function()
	Bubble.Visible = false
	Panel.Visible = true
end)

Minimize.MouseButton1Click:Connect(function()
	Panel.Visible = false
	Bubble.Visible = true
end)

QuestButton.MouseButton1Click:Connect(function()
	Status.Text = "📜 Quest selecionado"
end)

AttackButton.MouseButton1Click:Connect(function()
	Status.Text = "⚔ Ataque selecionado"
end)

TravelButton.MouseButton1Click:Connect(function()
	Status.Text = "🚢 Viagem selecionada"
end)

ConfigButton.MouseButton1Click:Connect(function()
	Status.Text = "⚙ Configuração"
end)
